rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // Aturan untuk koleksi users: 
    // User hanya boleh membaca dan menulis dokumen profil miliknya sendiri
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      // Mengizinkan user mengakses sub-collection di dalam profilnya (seperti produk, transaksi)
      match /{document=**} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }
    
    // (Opsional) Jika Anda masih menggunakan koleksi root lama, 
    // pastikan user yang login bisa mengaksesnya:
    match /products/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /product_categories/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /pos_transactions/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /refill_records/{document=**} {
      allow read, write: if request.auth != null;
    }
    match /app_settings/{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
